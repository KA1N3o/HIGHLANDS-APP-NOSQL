# Delete all promotions from Bigtable
# Run these commands in the Bigtable shell (cbt)

# Connect to Bigtable
cbt -project=highlands-coffee -instance=highlands-instance

# List all rows in promotions table to see what we have
read promotions

# deleteallrows promotions
